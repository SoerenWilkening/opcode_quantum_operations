### 1.1 `_sf_round_and_pack` — `softfloat_common.jl:198-227`, verbatim

```julia
@inline function _sf_round_and_pack(wr::UInt64, result_exp::Int64, result_sign::UInt64)
    # Overflow check
    exp_overflow = result_exp >= Int64(0x7FF)
    overflow_result = (result_sign << 63) | INF_BITS

    # Round to nearest even
    guard      = (wr >> 2) & UInt64(1)
    round_bit  = (wr >> 1) & UInt64(1)
    sticky_bit = wr & UInt64(1)
    frac       = (wr >> 3) & FRAC_MASK

    grs = (guard << 2) | (round_bit << 1) | sticky_bit
    round_up = (grs > UInt64(4)) | ((grs == UInt64(4)) & ((frac & UInt64(1)) != UInt64(0)))

    frac_rounded = frac + UInt64(1)
    mant_overflow = frac_rounded == IMPLICIT
    frac_final = ifelse(round_up,
                 ifelse(mant_overflow, UInt64(0), frac_rounded),
                 frac)
    exp_after_round = ifelse(round_up & mant_overflow,
                             result_exp + Int64(1),
                             result_exp)
    exp_overflow_after_round = exp_after_round >= Int64(0x7FF)

    # Pack normal result
    exp_pack = UInt64(clamp(exp_after_round, Int64(0), Int64(0x7FE)))
    normal_result = (result_sign << 63) | (exp_pack << 52) | frac_final

    return (normal_result, overflow_result, exp_overflow, exp_overflow_after_round)
end
```

**The docstring at `:193-197` names three returns and the function returns four.** It
reads *"-> (normal_result, exp_overflow, exp_overflow_after_round)"*
(`softfloat_common.jl:194`) while `:226` returns `(normal_result, overflow_result,
exp_overflow, exp_overflow_after_round)`. The **code** is the specification here; the
docstring is stale upstream and is quoted so nobody ports the tuple from it. Every one
of the five callers destructures four.

### 1.2 `_sf_normalize_clz` — `softfloat_common.jl:113-139`, verbatim

```julia
@inline function _sf_normalize_clz(wr::UInt64, result_exp::Int64)
    need32 = (wr & (UInt64(0xFFFFFFFF) << 24)) == UInt64(0)
    wr = ifelse(need32, wr << 32, wr)
    result_exp = ifelse(need32, result_exp - Int64(32), result_exp)

    need16 = (wr & (UInt64(0xFFFF) << 40)) == UInt64(0)
    wr = ifelse(need16, wr << 16, wr)
    result_exp = ifelse(need16, result_exp - Int64(16), result_exp)

    need8 = (wr & (UInt64(0xFF) << 48)) == UInt64(0)
    wr = ifelse(need8, wr << 8, wr)
    result_exp = ifelse(need8, result_exp - Int64(8), result_exp)

    need4 = (wr & (UInt64(0xF) << 52)) == UInt64(0)
    wr = ifelse(need4, wr << 4, wr)
    result_exp = ifelse(need4, result_exp - Int64(4), result_exp)

    need2 = (wr & (UInt64(0x3) << 54)) == UInt64(0)
    wr = ifelse(need2, wr << 2, wr)
    result_exp = ifelse(need2, result_exp - Int64(2), result_exp)

    need1 = (wr & (UInt64(1) << 55)) == UInt64(0)
    wr = ifelse(need1, wr << 1, wr)
    result_exp = ifelse(need1, result_exp - Int64(1), result_exp)

    return (wr, result_exp)
end
```

The target bit is **55**: the six masks are `0xFFFFFFFF << 24`, `0xFFFF << 40`,
`0xFF << 48`, `0xF << 52`, `0x3 << 54`, `1 << 55` — each a compile-time constant
(`:114`, `:118`, `:122`, `:126`, `:130`, `:134`), and each one's top bit is 55.

### 1.3 `_sf_handle_subnormal` — `softfloat_common.jl:174-191`, verbatim

```julia
@inline function _sf_handle_subnormal(wr::UInt64, result_exp::Int64, result_sign::UInt64)
    subnormal = result_exp <= Int64(0)
    shift_sub = Int64(1) - result_exp
    flush_to_zero = shift_sub >= Int64(56)
    shift_clamped = clamp(shift_sub, Int64(0), Int64(63))
    shift_u = UInt64(ifelse(flush_to_zero, Int64(0), shift_clamped))
    lost_mask_sub = (UInt64(1) << shift_u) - UInt64(1)
    lost_sub = ifelse((wr & lost_mask_sub) != UInt64(0), UInt64(1), UInt64(0))
    wr_sub_result = (wr >> shift_u) | lost_sub
    flushed_result = result_sign << 63

    wr = ifelse(subnormal,
         ifelse(flush_to_zero, wr, wr_sub_result),
         wr)
    result_exp = ifelse(subnormal, Int64(0), result_exp)

    return (wr, result_exp, flushed_result, subnormal, flush_to_zero)
end
```

**Its docstring (`:148-172`, `Bennett-xiqt` / U133) is upstream's record of a REFUSED
fix, and the port inherits the refusal rather than the question.** Upstream's own
review flagged `flush_to_zero = shift_sub >= 56` as theoretically dropping an RTNE
round-up; the disposition is *"investigated, doc-only"* (`softfloat_common.jl:166`),
pinned by `test/test_xiqt_subnormal_boundary.jl`. **Rule 1 settles what we do: port the
line.** It is quoted here because an implementer who reads `:177` without the docstring
will think it is a bug.

### 1.4 `_sf_normalize_to_bit52` — `softfloat_common.jl:68-105`, verbatim

```julia
@inline function _sf_normalize_to_bit52(m::UInt64, e::Int64)
    # Bennett-tpg0: defensive guard for m == 0. Substitute IMPLICIT
    # (1<<52, leading 1 already at the target bit) so the CLZ is a
    # no-op; restore zero outputs at the end.
    m_zero = m == UInt64(0)
    e_orig = e
    m = ifelse(m_zero, IMPLICIT, m)

    need32 = (m & (UInt64(0xFFFFFFFF) << 21)) == UInt64(0)
    m = ifelse(need32, m << 32, m)
    e = ifelse(need32, e - Int64(32), e)

    need16 = (m & (UInt64(0xFFFF) << 37)) == UInt64(0)
    m = ifelse(need16, m << 16, m)
    e = ifelse(need16, e - Int64(16), e)

    need8 = (m & (UInt64(0xFF) << 45)) == UInt64(0)
    m = ifelse(need8, m << 8, m)
    e = ifelse(need8, e - Int64(8), e)

    need4 = (m & (UInt64(0xF) << 49)) == UInt64(0)
    m = ifelse(need4, m << 4, m)
    e = ifelse(need4, e - Int64(4), e)

    need2 = (m & (UInt64(0x3) << 51)) == UInt64(0)
    m = ifelse(need2, m << 2, m)
    e = ifelse(need2, e - Int64(2), e)

    need1 = (m & (UInt64(1) << 52)) == UInt64(0)
    m = ifelse(need1, m << 1, m)
    e = ifelse(need1, e - Int64(1), e)

    # Restore zero: byte-identical for m != 0 (m_zero is false, ifelse
    # picks the CLZ result); for m == 0 returns (0, e_orig).
    m_final = ifelse(m_zero, UInt64(0), m)
    e_final = ifelse(m_zero, e_orig, e)
    return (m_final, e_final)
end
```

**Two structural facts, both from the source.** The target bit is **52**, not 55 —
the six masks are shifted by `21/37/45/49/51/52` against the other helper's
`24/40/48/52/54/55`, and the docstring says so in one line (`:65-66`, *"structure
mirrors `_sf_normalize_clz` but the target bit is 52 instead of 55"*). And the
`m == 0` guard is **post-`Bennett-tpg0`**: pre-tpg0 the function trusted callers and
returned `(0, e − 63)`; it now substitutes `IMPLICIT` and restores `(0, e_orig)`
(`:50-63`). Porting the pre-tpg0 shape is not available — it is not what is on disk —
but an implementer who "simplifies away" the guard as dead is re-deriving the
pre-tpg0 version, and the callers it breaks are every zero operand.

### 1.5 The constants — `softfloat_common.jl:8-14`

Identical to K22 §1.1, and already transcribed into `src/kernels/fpfield.h` as
`CQ_FP64_FRAC_MASK`, `CQ_FP64_IMPLICIT`, `CQ_FP64_EXP_MASK`, `CQ_FP64_INF_BITS`,

is a `const cq_bit[64]` SOURCE under PRD-v2 §7.3.

`CQ_FP64_EXP_ALL` (`= 0x7FF`) already exists in `fpfield.h`; `0x7FE` does not and is

twelve CLZ masks are `(0xFFFFFFFF << 24)` and friends **constant-folded**, which is
K15 phase J's reading and is what makes them views rather than shift rows.

### 1.7 The two encodings the hand-offs cross, and no C type expresses either

- **Into `_sf_round_and_pack` and `_sf_handle_subnormal`: the G/R/S working format.**
  `fadd.jl:13-14` states it — *"Working format: mantissa shifted left by 3 for
  guard/round/sticky bits. bit 55 = implicit 1, bits 54-3 = 52-bit fraction, bits 2/1/0
  = G/R/S."* `_sf_round_and_pack` reads exactly that at `:204-207`.
- **Into and out of `_sf_normalize_to_bit52`: leading 1 at bit 52.** `fdiv.jl:10-14`
  states why — the restoring loop *"assumes `ma, mb ∈ [2^52, 2^53)`"* — and
  `softfloat_common.jl:46-48` states the precondition (*"`m` has no bits set above bit
  52"*).

**A one-bit disagreement about either is right on every input that does not round and
wrong on the ties**, which is K15's risk 3 and is why the tie anchors (§5.10) are the
only detector. Five hand-off sites cross these two encodings and they cross a **bead**
boundary as well — `9ve.19` ↔ `9ve.21`/`9ve.22`/`9ve.25`/`9ve.26`/`9ve.27` — which is
where contracts go to be assumed. **Write the encoding into `fpround.h` as prose and
into the block struct's field names**; there is no type that carries it.

## 2. Gate sequence as an indexed block table

### 2.0 Nine decisions the transcription rests on

All nine follow PRD-v2 §7; six are places where §7's general rule needs a reading or
where two sibling K-docs disagree, and those are marked **[DERIVED]** and re-listed in
§6.6. **None may be settled silently in code** (CLAUDE.md, *Open blockers*).

**D-K23-1 — the grain is LITERAL and "the Julia type's width" is the rule.** PRD-v2
§7.2, one block per *"operator occurrence as the source spells it"*
(PRD-v2.md:620 @ 5e51ec1), as clarified by K18's D-K18-1: a `Bool` intermediate is a **one-bit** span
and a `UInt64`/`Int64` intermediate is **sixty-four**. In M32 the `Bool`s are
`m_zero`, the twelve `need*`, `subnormal`, `flush_to_zero`, `exp_overflow`,
`round_up`, `mant_overflow`, `exp_overflow_after_round` and the four clamp predicates;
everything else is 64 bits, including `exp_pack` (11 bits of value) and `result_sign`
(1 bit of value). **No narrowing towards the field width**, which is D9's K12
precedent.

**D-K23-2 — `Int64` and `UInt64` are the same 64 `cq_bit`s, and the reinterpret is
wiring. [DERIVED — CONFIRMED 2026-09-18 at implementation.]** `shift_u` (`:179`),
`exp_pack` (`:223`) and `e_orig = e` (`:73`) are `CQ_FROP_VIEW` rows with shift 0 and
an all-ones mask: zero slots, zero bits, and the only thing they change is which
comparator the next row reaches for. `UInt64(ifelse(...))` at `:179` and `UInt64(clamp(...))` at `:223`
are **views**: zero gates, zero qubits. What changes is which comparator the *next* row
uses, and §5.4 is the table of which. K15 §2.0 fixes the same reading for M33.

**D-K23-3 — a compile-time-constant mask or shift is a VIEW.** PRD-v2 §7.3 as amended
2026-09-18, *"COMPILE-TIME CONSTANT is WIRING, not a block"*
(PRD-v2.md:644 @ 5e51ec1). This is worth **41 of M32's 154 rows** (§2.2's tally) and it is the reason
this document's counts differ from K15 phase J/K's and from K21 §5.3's — both drafted
under the and-block reading. §5.9 reconciles them arithmetically rather than by
assertion. **A RUNTIME mask is still a block**: `wr & lost_mask_sub`
(`softfloat_common.jl:181`) is the only `&` in M32 whose second operand is computed,
and it is `C_and(64)`.

**D-K23-4 — the `==` polarity is `lower_not1!`, a fresh wire and two slots.
[DERIVED, and the two sibling drafts disagree. CONFIRMED 2026-09-18: M32 SHIPPED
M31's spelling — twenty `lower_not1!` rows, 40 slots and 20 bits across the four
helpers, fifteen of them for an `==` and five for the polarity of an ordered
compare. K15's in-place `X` remains unadopted and the reconciliation remains a
K-doc question, not a code one.]** M16's `eq` block's raw wire is
`a != b` (`cmp.h`, *"THE RAW FLAG IS `a != b`"*), so a `!=` occurrence reads it and an
`==` occurrence owes itself a polarity. **K22 §2.0 spends one `lower_not1!`
(`arith.jl:474-478`) — CX into a fresh bit, then X — and M31 SHIPPED that way**; K15
§2.2 instead writes `== is C_eq(W) + 1`, *one* X flipping the block's own raw flag in
place. The two differ by one slot and one bit per occurrence, **fifteen occurrences in
M32**, and K22 §5.3 records the in-place form as *"not adopted: it is not in
Bennett"*. This document follows **M31's shipped choice**, because M32 is the module
every other fp kernel composes and two spellings of the same polarity in one library is
the thing D9(b) refused for `uge`. **It is a cost fork, not a correctness fork** — and
it is a PRD-v2 §7.2 or K-doc-level reconciliation, not a kernel choice.

**D-K23-5 — `clamp(x, lo, hi)` is transcribed as `ifelse(x < lo, lo, ifelse(x > hi,
hi, x))`, four rows. [DERIVED — the body is NOT in the snapshot, §1.6. CONFIRMED
2026-09-18 as the reading SHIPPED, at both sites and in the same row order, in
`src/kernels/fpround.c` and again in `cq_handle_subnormal_eval`'s `clamp_i64`. And
a finding the draft did not have: at `:178` the clamp is **value-inert on every
reachable input** — `shift_clamped` reaches `shift_u` only when `subnormal` is true
and `flush_to_zero` false, so `shift_sub` is in [1, 55] and both arms are the
identity there. So no L1 comparison can see a mis-ordered or mis-signed clamp at
THAT site; the slot scan and L4 are its only detectors, and `:223`'s clamp — whose
`exp_pack` reaches `normal_result` — is where L1 does see one.]** The rows are
`slt`(+polarity), `sgt`(+polarity), `mux`(hi), `mux`(lo). **Any reading agrees in value
wherever `lo <= hi`**, which both M32 sites satisfy by construction (`0 <= 63`,
`0 <= 0x7FE`), so the fork is in the **row order and the cost**, not the answer — and
the order is what a slot scan pins. K15 §2.2's phase K rows K4–K7 already chose this
order; M32 must not choose the other one, and if the maintainer prefers a different
ordering it changes **both** documents. **Do not "check it against the source": there
is no source** — that is precisely the gap.

**D-K23-6 — the composition identity is over SLOTS, never gates.** K18's D-K18-6, *"the
composition identity is over SLOTS, not gates"* (K18.md:392 @ 5e51ec1), and K22 §2 says
M31 is the first module where it bites. M32 is the second, and it bites harder: 41 view
rows put constant lanes *inside* block operands at every mask, so a compute slot may
emit **nothing** or emit a `CX` where the schedule says `CCX`. **Do not "fix" a red slot
scan by asserting one-gate-per-slot.**

**D-K23-7 — the same block costs different GATES in different CALLERS, and that is new
here. [DERIVED — CONFIRMED AND MEASURED 2026-09-18: §3.4's table pins two rows per
block and the caller-shaped one is strictly cheaper in all four, by 309 / 247 / 248
/ 219 gates. `tests/goldens/fpround.counts` carries `fwd@allq` and `fwd@caller`, and
`the_caller_shaped_mask_is_cheaper_and_the_alphabet_is_x_cx_ccx` asserts the
inequality reading no golden — an implementation that MATERIALISED its constants
would collapse the two onto one number.]** M31's views fold identically at every operand mask *within one
kernel*. M32's do not: `result_sign` is a computed 64-bit span in `fadd`/`fmul`/`fdiv`/
`fma` and the **literal `UInt64(0)`** in `fsqrt` (`fsqrt.jl:103`), and `e` is a computed
span in `fmul`/`fdiv`/`fma`/`fsqrt` and the **literal `Int64(1)`** in `flog.jl:269`,
`fpconv.jl:75` and `fpow.jl:758`. A constant argument makes `result_sign << 63` an
all-`CQ_BIT_ZERO` view, so every `or` reading it folds. **Consequence for the goldens:
M32's L4 row is per (block, caller-shaped operand mask), not per block** — the same
trap K16's finding records for `n_a·n_b` inside M34, in a new place. Slots are
unaffected.

**D-K23-8 — a returned VIEW has no home in the caller's region, so its accessor
ASSEMBLES it. [DERIVED — CONFIRMED 2026-09-18: `cq_subnorm_flushed(k, out[64])` fills
a caller array and is the only accessor in M32 that does. A rider the draft did not
have: because row 18 is an operand of NOTHING, `result_sign` is never resolved while
STEPPING `_sf_handle_subnormal` — so that block's NULL-input guard is unreachable
through its own step function and only `cq_subnorm_flushed` reaches it. Measured, by
a death case that did not abort.]** `flushed_result = result_sign << 63`
(`softfloat_common.jl:183`) is a view over an operand, not a scratch span, so
`cq_subnorm_flushed()` cannot return a `const cq_bit *` into the region the way
`cq_eq_flag` does. It fills a caller-supplied `cq_bit[64]`, exactly as
`cq_fp_view_exp` does (`fpfield.h`). Materialising it into the region instead would
cost 64 bits and a `W`-gate copy to buy a pointer — a block emitting a copy nothing
reads, which is the shape §7.3 exists to refuse.

### 2.1 Vocabulary

`W = 64` throughout (PRD-v2 §1: `f64` only). Three kinds of thing appear as a block
operand, exactly as in K18 §2.1: a **caller span** (M32's blocks have no rails — every
input arrives as a span the consumer owns), a **scratch span** pre-materialised by
`cq_sandwich` at step 0 under I6(b), and a **constant span or view**, which owns no
qubits.

The constant spans M32 needs, each a `const cq_bit[64]` built by `cq_fp_const`
(`fpfield.h`) and owning nothing:

```
K_ZERO   all 64 lanes CQ_BIT_ZERO       — every `== 0` / `!= 0` operand and every fill
K_ONE    lane 0 ONE                     — `+ 1`, `- 1`, `1 - x`, `ifelse(.., 1, 0)`
K_FOUR   lane 2 ONE                     — `grs > 4`, `grs == 4`
K_56  K_63  K_7FE  K_7FF                — the four thresholds
K_D32 K_D16 K_D8 K_D4 K_D2 K_D1         — the six CLZ decrements
K_IMPLICIT  K_INF  K_FRAC               — from fpfield.h
KC52_k  ×6   KC55_k  ×6                 — the twelve constant-folded CLZ masks
```

**The twelve CLZ masks are constants and NOT shift rows**, which is K15 phase J's
reading of `(UInt64(0xFFFFFFFF) << 24)`: the inner shift's operands are both literal,
so Julia folds it and there is no operator occurrence left to transcribe. The *outer*
`m & KC` is D-K23-3's view. **`K_FOUR` and `K_D4` carry the same bit pattern and are
two spans**, because §7.2's no-CSE rule is about occurrences; naming them apart is
free and makes the slot scan's constant-lane accounting readable.

### 2.3 The slot scan — written BEFORE the kernel

**Four-valued, with its own stream cursor**, exactly as K22 §2.3 and K18 §2.7. The
prediction per slot is a *kind or nothing*, because a `CQ_BIT_ZERO` control makes the
gate vanish and a `CQ_BIT_ONE` demotes it (`CCX→CX`, `CX→X`), and both come from the
views and the constant spans rather than from operand bit-kinds — so they happen at
**every** mask, including the one L4 pins.

```
/* the per-block op-kind schedules, each from the pinned loop it ports */
sched_eq(n, v)  = v < 2n ? CX : (((v - 2n) % 3 == 2) ? CCX : CX)    /* arith.jl:424-447 */
sched_ult(n, v) = ...                                               /* arith.jl:449-463 */
sched_slt(n, v) = ...                                               /* arith.jl:465-472 */
sched_sub(n, v) = ...                                               /* adder.jl:148-172 */
sched_add(n, v) = ...                                               /* adder.jl:1-18    */
sched_mux(v)    = (v % 4 == 3) ? CCX : CX                           /* arith.jl:522-532 */
sched_or(v)     = (v % 3 == 2) ? CCX : CX                           /* arith.jl:274-282 */
sched_and(v)    = CCX                                               /* arith.jl:268-272 */
sched_not1(v)   = (v == 0) ? CX : X                                 /* arith.jl:474-478 */
sched_barrel(...) = ...                                             /* arith.jl:350-400 */

predict(slot s):
    r    = the row of §2.2 containing s;  v = s - base(r)   /* base ASKED, never written */
    kind = sched_<r.block>(r.n, v)
    for each CONTROL lane c the gate reads:
        if c is a constant ZERO -> return NONE
        if c is a constant ONE  -> kind--                   /* CCX->CX, CX->X */
    return kind
```

`CQ_OP_X`/`CX`/`CCX` are 0/1/2, so "demote" is a decrement, and the test pins that
numbering with a `_Static_assert` rather than a comment — `CQ_BIT_ZERO`'s and
`CQ_ANGLE_GENERAL`'s precedent.

**Four things this scan must get right that K22's did not have to.**

1. **Three of the schedules ALREADY EXIST and are reused verbatim; the rest are
   transcribed once and SHARED.** Measured: `tests/test_kernel_divrem_schedule.inc`
   encodes `sched_ult`, `sched_sub` and `sched_mux` and nothing else, so `sched_eq`,
   `sched_slt`, `sched_add`, `sched_barrel`, `sched_and`, `sched_or` and `sched_not1`
   are new — and must live in **one** shared place, because K18, K22, K17 and every
   later fp kernel need the same ones. K18 §2.2 records that reusing a schedule *"is
   the whole point of composing a shipped block"*; a second transcription of one is a
   second chance at the same off-by-one.
2. **Non-vacuity is asserted per helper and EXACTLY, never as `demoted > 0`.** K22
   §2.3 measured that a blanket inequality goes red on the classes whose constant is
   zero. Here the sharp claim is per row: `folded` counts the ZERO lanes of that row's
   operands and `demoted` counts the ONE lanes, both derivable from the constant's
   `popcount` and from the view's lane map.
3. **The prediction must be taken per CALLER-shaped operand mask** (D-K23-7), because a
   constant `result_sign` collapses R3, R41 and H19's neighbourhood and a constant `e`
   collapses every `sub` in the CLZ ladders. At minimum the scan runs at two shapes:
   all-spans-quantum, and `fsqrt`'s (`result_sign` all-`CQ_BIT_ZERO`).
4. **A stage's mask view folds a KNOWN number of lanes, and that number is the
   stage's own.** `KC52_32` leaves 32 lanes, `KC52_1` leaves one; so `S*k*2`'s `eq`
   block sees `64 − popcount(mask)` constant-ZERO lanes and its Phase-A copies elide
   accordingly. **Phase B of an `eq` block never folds** — its controls are `diff` and
   `orr` bits, pre-materialised at step 0 under I6(b) — which is K22 §2.3's finding and
   holds here for the identical reason.

### 2.3a The span-disjointness scan

**K22 §2.3a exists because a mutant survived everything else**: collapsing two `eq`
blocks onto one offset left the value right, the palindrome intact, the pool restored,
every gate count identical and the slot scan green (K22.md:204 @ 5e51ec1). M32 has
**113 emitting rows with their own extents**, against M31's five spans, so the same
instrument is mandatory and is the thing to copy — **not** K22's argument about why the
collapse happened to be value-equivalent there, which was an accident of `lower_eq!`'s
zero-on-equal behaviour and does not transfer.

The scan walks the same slot→stream cursor and asserts that **no qubit index is written
by two of §2.4's spans**. It reads no golden and is derived from §2.2's table.

**And M32 needs a second layer of it that M31 did not: the same block type appears up
to fifteen times.** An off-by-one in the per-row offset table lands inside a *different
instance of the same block*, so the gates are plausible, the palindrome survives and
only the value moves. That is K15's risk 1 ("172 boundaries where K12 has four") at 154
boundaries, and it is why §2.4's offset table is a compile-time prefix sum rather than
arithmetic at the call site.

is two blocks at two offsets in one region.

```
S_norm52  = 7·S_eq  + 7·1 + 15·S_mux + 6·S_sub
S_clz     = 6·S_eq  + 6·1 + 12·S_mux + 6·S_sub
S_subnorm = 4·S_slt + 2·1 +  2·S_sub + 7·S_mux + 64 + S_eq + 64 + 2·S_barrel
S_round   = 4·S_slt + S_ult + 5·1 + 3·S_eq + 2·S_add + 5·S_mux + 5·64 + 2·1 + 1
```

every `S` **asked of its module at width 64**, never written down. §4 carries an
illustrative substitution and says why it is not a figure to pin.

### 2.5 There is no copy-out — M32's blocks have OUTPUTS, not results

A Rule 7 kernel ends with `dst ^= result` and `cq_sandwich` runs the copy-out with the
I6 extent disarmed. **M32 ships no kernel**, so none of its blocks copies anything out:
each publishes its outputs as accessors into the caller's region, and the *consumer's*
sandwich does the one copy-out at the end of the whole fp kernel. That is M12's,
M18's and M19's shape (`cq_barrel_result`, `cq_mul_product`, `cq_divrem_quotient`),
and it is why §2.6's I6(a) discharge is complete without naming `dst` at all.

### 2.6 I6(a) and I6(b)

**I6(a) — every target a block names is a bit of the CALLER's region. Yes, by
construction.** Every row of §2.2 is a view (emits nothing) or a block whose targets
are that block's own vectors at `off + <span>`. `wr`, `result_exp`, `result_sign`,
`m` and `e` — M32's five inputs — appear **only ever as `const cq_bit *` controls**,
which `cq_emit_*`'s signature makes structural (Rule 8's first enforcement mechanism).
The constant tables are `const` for the same reason.

**I6(b) — every scratch bit is `CQ_BIT_Q` for the whole compute half. Yes, and it is
the consumer's driver's doing**, since M32's blocks are never a sandwich of their own.
Two consequences, both already stated for K15: no fold ever fires on a scratch bit, so
every gate whose controls are all scratch emits; and **the folds that DO fire are
exactly the constant-source and view ones**, identically forward and backward, so a
constant-elided slot is still a slot and the palindrome is unaffected.

**An input may be a view that overlaps a span an earlier row wrote**, which is the
sanctioned calling shape (plan §0.4 obligations 3 and 4; K12 §2.1a). M35 hands
`_sf_normalize_to_bit52` an `ma` that is its own phase-E mux output, and hands
`_sf_round_and_pack` a `wr` that is `_sf_handle_subnormal`'s output. Guards compare
**ranges**, never base pointers.

### 2.7 The export shape

Four blocks, all on the shape `cq_divrem_block` / `cq_mul_block` / `cq_barrel_block`
already have — **`cq_scratch *scr` plus a `uint32_t off`, and a `cq_*_region()`** —
rather than `cq_ult_block`'s named-vector shape, because each of these has twenty to
forty internal spans and naming them in the struct would put M32's layout in its
consumers' headers.

```c
typedef struct {
    const cq_bit *wr, *result_exp, *result_sign;  /* 64 lanes each; CONTROLS ONLY */
    cq_scratch   *scr;      /* the CALLER's region — the block allocates nothing  */
    uint32_t      off;      /* where this block's cq_round_region() bits start    */
} cq_round_block;

uint32_t cq_round_region(void);
int      cq_round_steps (void);
void     cq_round_step  (cq_ctx *ctx, const cq_round_block *k, int u);

const cq_bit *cq_round_normal            (const cq_round_block *k);  /* 64 */
const cq_bit *cq_round_overflow_result   (const cq_round_block *k);  /* 64 */
const cq_bit *cq_round_exp_overflow      (const cq_round_block *k);  /*  1 */
const cq_bit *cq_round_exp_overflow_after(const cq_round_block *k);  /*  1 */
```

and the same shape for `cq_clz_block`, `cq_subnorm_block` and `cq_norm52_block`, with

- `cq_subnorm_flushed(k, cq_bit out[64])` **filling** a caller array rather than
  returning a pointer (D-K23-8),
- `cq_subnorm_flag()` and `cq_subnorm_ftz()` returning the two raw compare wires,
- `cq_norm52_m()` / `cq_norm52_e()`,
- `cq_clz_wr()` / `cq_clz_exp()`.

**No width parameter anywhere**, for M31's reason: PRD-v2 §1 scopes v2 to `f64`, so 64
is a constant and not a ladder (`fpfield.h`, *"there is no `W` to be generic over"*).
`cq_fp_class_region` returns `uint32_t` where M12/M18/M19 return `int`; M32 follows
**M31**, since it is the fp neighbour and a consumer's offset arithmetic is unsigned.

**Plus the four classical rows (D-K23-9):**

```c
void cq_round_and_pack_eval(uint64_t wr, int64_t rexp, uint64_t rsign,
                            uint64_t *normal, uint64_t *ovf,
                            int *exp_ovf, int *exp_ovf_after);
void cq_normalize_clz_eval (uint64_t *wr, int64_t *rexp);
void cq_handle_subnormal_eval(uint64_t *wr, int64_t *rexp, uint64_t rsign,

void cq_normalize_to_bit52_eval(uint64_t *m, int64_t *e);
```

**and there is NO R9 short-circuit inside a block**, which is not an oversight: `mul.h`
already states it — a block *is* the compute half and its region is allocated and
pre-materialised by the time step 0 runs, so there is nothing left to fold. **M32's
fold belongs at each consumer's entry, over the whole routine**, and the four `_eval`
functions above are what those entries call.

> ### THE IDENTITY IS THE ASSERTION THAT SURVIVES `CQOPS_UPDATE_GOLDENS`
>
> It reads **no golden**, so `CQOPS_UPDATE_GOLDENS=1` cannot bless a reduction — and
> M32 offers the reduction on a plate. `exp_pack` is 11 bits in a 64-bit span; `grs` is
> 3; `guard`, `round_bit`, `sticky_bit` and every `need_k` are 1; `frac` is 52. **The
> "shorten it" mutant is available on essentially every span**, it looks like an
> optimisation every time, and L1/L2/L3/L5 stay green for all of them because the
> values are right. PRD-v2 §7.2 refuses the narrowing on the ground D9 refused K12's.
> **Build this check before writing M32.**
>
> It lives in **two** cases, as K12's and K15's do: one that runs **no block** and pins
> the eight `C_*` symbols against their modules, and one that pins each of M32's four
> against what those just measured. A sibling's cost moving then reddens the first and
> **names the block**.

## 4. Ancilla count

**M32's blocks allocate nothing** (plan §0.4 obligation 2). The region is the
consumer's, and the figure that matters to a consumer is §2.4's `S_*`, which is
**~1.6 × 10⁴ bits for all four** on §3.2's illustrative substitution — on the order of
`S_fadd`'s own 21k (K15 §2.1's hand-count) and therefore **not** a rounding error in
anyone's budget.

**Nothing in M32 may be freed mid-compute-half.** `wr_lost`, the discarded CLZ stage
outputs, `overflow_result` on every caller that ignores it (`fdiv.jl:122`'s `_`) and
the whole `frac_rounded` span when `round_up` is false are all provably dead
*mathematically* and none is provably `|0⟩` to the two-bit shadow. The consumer's
sandwich reverse is the only thing that returns them. K12.md and K15.md carry the
identical warning.

**R11's pool ceiling.** M32 is a fixed additive cost on five kernels, three of which
(`fdiv`, `fma`, `fsqrt`) call it two or three times. The disposition is already decided
(PRD-v2 §4's R11, PRD §15 D25) and the live prohibition is that **neither M32 nor the
shim may grow its own "that will not fit" check**: `cq_qubits_acquire` fails hard
naming the pool, at the pre-materialisation site, before a gate is emitted.

### 5.9 No CSE — the repeated sub-expressions, per occurrence

PRD-v2 §7.2 forbids sharing, and M32 repeats more than any module so far:

- **`result_sign << 63`** — twice in `_sf_round_and_pack` (R2, R39) and once in
  `_sf_handle_subnormal` (H19). Three views, zero cost, three rows.
- **`result_exp >= Int64(0x7FF)`** (R1) and `exp_after_round >= Int64(0x7FF)` (R31) —
  the *same predicate on different operands*, so not a CSE candidate at all, but they
  share a constant span and an implementer will be tempted to share the block.
- **The clamp** — H4–H9 and R32–R37 are the same four-row shape at two sites with
  different bounds. **Two instances, two extents**; §2.3a is what catches a collapse.
- **The six CLZ stages ×2 helpers** — twelve instances of one seven-row shape. One C
  body with a mask/target-bit table is fine (§2.2.2); one *instance* is not.
- **`frac & UInt64(1)`** (R19) against `wr & UInt64(1)` (R8) — different operands, both
  views.

**And the reconciliation with the two drafts that counted these differently.** K21 §5.3
gives `_sf_normalize_to_bit52` as *"41 rows, 34 emitting, scratch `6 × 639 + 255 + 256
= 4,345` bits"*. This document gives 48 rows, 35 emitting, `S_norm52` ≈ 3,968 bits. The
difference is **exactly two bookkeeping changes and no disagreement about the
construction**: (i) PRD-v2 §7.3's amendment turns the six `and`(64) mask blocks into
views, removing `6 × 64 = 384` bits and six emitting rows; (ii) D-K23-4 gives each `==`
polarity its own `lower_not1!` row and its own bit, adding seven rows and `7` bits.
`4,345 − 384 + 7 = 3,968`. **Arithmetic, not assertion** — and it is the check to
re-run if either decision moves.

### 5.10 L1's anchors for M32 — PRD-v2 §7.12, refined against these four sources

§7.12 requires the list be refined against *"the source's own special-case
predicates"*: **each named predicate gets an anchor that makes it true**, because each
is a branch of the `ifelse` tree a random draw of 32 reaches with no guarantee
(PRD-v2.md:863 @ 5e51ec1). M32's named predicates are `m_zero`, the twelve `need_k`,
`subnormal`, `flush_to_zero`, `round_up`, `mant_overflow`, `exp_overflow`,
`exp_overflow_after_round` and the four clamp predicates.

| # | predicate it makes true | input shape (`wr`, `result_exp`, `result_sign`) or (`m`, `e`) | what it catches |
|---:|---|---|---|
| 1 | `round_up` false, `grs = 0` | `wr` with bits 2/1/0 clear | the no-rounding baseline |
| 2 | `round_up` via `grs > 4` | G set, R or S set | R15's polarity |
| 3 | **tie to EVEN, rounds DOWN** | `grs == 4` exactly (G set, R and S clear) and `frac` bit 0 **clear** | R17/R21's conjunction |
| 4 | **tie to ODD, rounds UP** | `grs == 4` exactly and `frac` bit 0 **set** | the other half of the tie |
| 5 | `mant_overflow` | `frac == FRAC_MASK` with `round_up` | R23/R24, the subnormal→normal carry |
| 6 | `subnormal`, no flush | `result_exp` in `[−54, 0]` | H1, and the SIGNED compare |
| 7 | `flush_to_zero` | `result_exp <= −55` so `shift_sub >= 56` | H3; **live in `fdiv`/`fmul`/`fma` and DEAD in `fadd`** |
| 8 | `exp_overflow` | `result_exp >= 0x7FF` | R1 |
| 9 | `exp_overflow_after_round` **without** `exp_overflow` | `result_exp == 0x7FE` and a carry out of `frac` | R28–R31; the only row that separates the two flags |
| 10 | every clamp arm | `result_exp` below 0 and above `0x7FE`; `shift_sub` below 0 and above 63 | H4/H6/R32/R34 |
| 11 | `m_zero` | `m == 0` | `_sf_normalize_to_bit52`'s tpg0 guard |
| 12 | each `need_k` alone, and the largest reachable combination | `m`/`wr` with its leading 1 at the target bit (no stage fires), at bit 0 (the maximum shift) and at each intermediate position | the twelve CLZ stages, one per mask |
| 13 | `result_sign` constant `0` | `fsqrt`'s call shape | D-K23-7's per-caller fold |
| 14 | `e` constant `Int64(1)` | `flog`/`fpconv`/`fpow`'s call shape | the same, on the other argument |

**Rows 3, 4 and 9 are the only detectors of §1.7's encoding contract**, and a random
draw reaches them essentially never. **Row 7 is DEAD in `fadd`** — K15's finding, from
upstream's own `Bennett-xiqt` measurement — so an anchor list that claims it for every
caller lists a row that cannot go red in one of them; state it per caller.

> **ALL SIX CLZ STAGES CAN NEVER FIRE TOGETHER, IN EITHER HELPER — so an anchor row
> demanding it is a row that cannot go red.** The stages that fire are exactly the
> binary representation of the shift count `s`, and `s` is bounded by the distance from
> lane 0 to the target bit: `s ∈ [0, 52]` in `_sf_normalize_to_bit52` and `s ∈ [0, 55]`
> in `_sf_normalize_clz`, while all six stages together are `32+16+8+4+2+1 = 63`. The
> largest reachable popcount is **five** in both — `s = 31` gives `16+8+4+2+1` and
> `s = 47` gives `32+8+4+2+1`, both in range for either helper, while the six-stage
> `s = 63` is out of range for both. Row 12 therefore asks for *each stage individually* plus the
> largest reachable combination, and **the implementer computes which that is per
> helper rather than assuming the two are the same.** This is a claim about the port's
> reachable inputs, hand-derived here and UNVERIFIED-BY-EXECUTION like everything else
> in this document.

**And an anchor here is not a `double`.** M32's inputs are a working-format `wr` and an
unbiased `result_exp`, not an IEEE pattern, so §7.12's `±0`/`±Inf`/NaN anchors belong
to the **consumers** and reach M32 only through their select chains. That is the whole
of §6.5's risk 1.

### 5.11 What is dead in which caller — three rows, three different reasons

Recorded because each will look like a bug to whoever ports the consumer, and because
**§7.6's discarded-arm rule says all three are emitted anyway**: *"A DISCARDED ARM IS
STILL EMITTED, WHICH IS §7.2's GRAIN AND NOT A COST TO OPTIMISE AWAY"*
(PRD-v2.md:757 @ 5e51ec1).

- **`overflow_result` (R3) is dead in `fdiv`.** `fdiv.jl:118-121` says so in its own
  words (`Bennett-ardf` / U138) and destructures it as `_` at `:122`. It is **live** in
  `fadd`, `fmul` and `fma`, whose select chains use it.
- **`flush_to_zero` (H3) is dead in `fadd`** — §5.10 row 7.
- **The tie branch (R17/R21) is dead in `fsqrt`** — Kahan's theorem, which upstream
  states at `fsqrt.jl:11-15` and K21 §5.5 records. `fsqrt` therefore provides **no**
  witness for anchors 3, 4, 5, 7, 8 or 9.

**None of the three may be narrowed, skipped or folded.** Narrowing strands the mux
that selects it and is a Rule 1 re-derivation.

**TWO OF THE FIFTEEN KILLED BY ABORTING THE BINARY AND PRINTED NO `not ok` LINE**,
which is the recorded exit-code trap reproduced live: a battery grepping only for
`^not ok` reports them as survivors. Both were caught because the runner also reads
the binary's exit code.

**THE TWO SURVIVORS ARE A PROVED EQUIVALENT CLASS AND THE PAIRED MUTATION IS THE
PROOF.** Exchanging the two operands of a COMMUTATIVE row — `lower_and!`'s Toffoli,
`lower_or!`'s first two CNOTs — changes no value, no target, no gate tuple, no
palindrome and no scratch. Exchanging them on a row whose operator is NOT
commutative, `_sf_handle_subnormal`'s `Int64(1) - result_exp`, is killed by four
cases. So the survival is a fact about commutativity rather than a hole, and it is
recorded at the site in `fpround_emit.c` with the instruction not to "simplify"
either line to match the mutant — `lower_or!`'s order is `arith.jl:276-278`'s and
Rule 1 is about provenance, not only about behaviour.

**WHAT IT DOES EXPOSE, FILED RATHER THAN BUILT: the slot scan predicts an op KIND
per slot and advances its cursor only on a non-NONE, so it cannot see WHICH operand
a slot reads.** Exchanging which of two operands folds moves the fold from one slot
of a group to another and the cursor counts the same one gate. Every instrument in
this repo that walks a recorded stream checks TARGETS (the span scan) or KINDS (the
slot scan); none checks CONTROL indices. For M32 that is covered by L1 wherever the
operator is not commutative, which is every row that matters. K15, K16, K17, K20 and
K21 inherit the same instrument set and the same blind spot.

1. **M32 HAS NO HOST ORACLE — DECIDED 2026-09-18 AS PRD-v2 §7.16.** L1 is a
   hand-written IEEE-754 reference model in `tests/support/fpref.[ch]`, written from
   the standard's rules and never from the Julia; the kernel-level host differential
   at M33/M35 is the end-to-end check on top. `src/kernels/fpround_eval.c` is the
   second C transcription this risk warned about, and it is CHECKED AGAINST the
   model rather than used as one. The independence is spelled out in `fpref.h` so a
   later reader can check it rather than take it: the model's rounding decision is
   `G && ((R|S) || odd)` where the port packs `grs` and compares against 4, its
   normalisation is a counting loop where the port is a six-stage binary search, and
   its subnormal shift is one place at a time where the port clamps, muxes and
   barrel-shifts. **Measured**: `_eval` rounding ties away instead of to even is
   killed by the model and by nothing else.

> **THE ORIGINAL STATEMENT OF RISK 1, KEPT SO THE DECISION CAN BE READ AGAINST
> WHAT IT ANSWERED.** M32 HAD NO HOST ORACLE. There is no C
   `_sf_round_and_pack` to differential against, and PRD-v2 §7.4's own instruction —
   *"L1's oracle stays independent: the host operator, with the IEEE-unspecified cells
   pinned by TABLE"* — names an operator that does not exist at this layer. The three
   candidates are (a) a hand-written `(wr, result_exp, result_sign) → outputs` table
   from §5.10's anchors, (b) a **second, independently written** C transcription in
   `tests/support/`, and (c) deferring the real differential to M33/M35's host
   comparison. **(b) alone is the Step 18 trap** — an oracle sharing code with the
   implementation is blind to exactly what that code gets wrong — and **(c) alone means
   M32 ships untested against anything.** This needs a decision before `9ve.19` starts,
   not an implementer's judgement.
2. **THE G/R/S AND BIT-52 ENCODINGS (§1.7).** Five hand-off sites, two encodings, no C
   type, and a bead boundary at every one. Wrong only on ties — anchors 3, 4, 9.
3. **THE EIGHT SIGNED COMPARES (§5.4).** Invisible to the slot scan, to L4 and to the
   palindrome; only L1 with an underflow anchor sees it. And `9ve.31` does not exist
   yet, so an implementer reaching for `cq_ult_block` mid-kernel is reaching for the
   silently wrong substitute.
4. **THE SLOT ARITHMETIC — 154 boundaries.** §2.3's scan, four-valued and with its own
   stream cursor, plus §2.3a's disjointness scan. The twelve CLZ stages are the one
   repeating structure, so an off-by-one there is absorbed by the next stage and
   produces a plausible gate rather than a crash. **Write both scans before the
   module.**
5. **`clamp` IS NOT A PORT (§1.6, §5.5).** Two sites, four rows each, no source to
   check against, and K15 already carries the identical gap unresolved.
6. **THE `==` POLARITY FORK (§5.6) AND THE CONSTANT-MASK AMENDMENT (§5.9).** Both are
   cost forks between documents rather than correctness forks — and both are worth
   hundreds of slots and hundreds of qubits across five kernels, and both are one
   silent code edit away from being settled the wrong way for the whole port.
7. **THE RECORDED SEAM (§5.8).** Rule 12 says a split is scheduled, never a surprise;
   the schedule on record predates the module tripling in size.
8. **THE TWO MISSING EXPORTS (§5.7).** `cq_*_region()` for six blocks and
   `cq_mux_steps(W)`. Not a correctness risk; a "write the number down" risk, which is
   the one this project's composition checks exist to remove.
